  <script>
    function openPamanaViewer() {

      const normalViewer =
        document.getElementById('pamana-normal-viewer');

      const fullscreenViewer =
        document.getElementById('pamana-fullscreen');

      const fullscreenIframe =
        document.getElementById('pamana-fullscreen-iframe');

      /*
       * Remove the normal Heyzine iframe.
       * This stops its audio and ensures only one
       * Heyzine viewer is active.
       */
      normalViewer.innerHTML = '';

      /*
       * Load the Heyzine reader into the fullscreen-style
       * overlay only after the button is clicked.
       */
      fullscreenIframe.src =
        'https://heyzine.com/flip-book/a61f59994b.html';

      fullscreenViewer.style.display = 'block';

      document.body.style.overflow = 'hidden';
    }

    function closePamanaViewer() {

      const normalViewer =
        document.getElementById('pamana-normal-viewer');

      const fullscreenViewer =
        document.getElementById('pamana-fullscreen');

      const fullscreenIframe =
        document.getElementById('pamana-fullscreen-iframe');

      /*
       * Remove the fullscreen Heyzine iframe.
       * This stops its audio.
       */
      fullscreenIframe.src = '';

      fullscreenViewer.style.display = 'none';

      /*
       * Restore the normal Heyzine viewer.
       */
      normalViewer.innerHTML = `
        <iframe
          src="https://heyzine.com/flip-book/a61f59994b.html"
          title="The All Star Annual 2026–2027 Edition"
          allowfullscreen
          allow="autoplay; fullscreen; clipboard-write"
          loading="lazy"
          style="
            display:block;
            width:100%;
            height:600px;
            border:1px solid lightgray;
          "
        ></iframe>
      `;

      document.body.style.overflow = '';
    }
  </script>
